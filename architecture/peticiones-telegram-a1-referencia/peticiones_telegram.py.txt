def leer_env(ruta=os.path.join(CASA, ".starseed", "env")):
    datos = {}
    try:
        with open(ruta) as f:
            for linea in f:
                linea = linea.strip()
                if not linea or linea.startswith("#") or "=" not in linea:
                    continue
                k, v = linea.split("=", 1)
                datos[k.replace("export ", "").strip()] = v.strip().strip('"').strip("'")
    except OSError:
        pass
    return datos

def log(*partes):
    texto = " ".join(str(p) for p in partes)
    if TOKEN:
        texto = texto.replace(TOKEN, "***")
    print(datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S"), texto, flush=True)


def leer_json(ruta, defecto):
    try:
        with open(ruta) as f:
            return json.load(f)
    except (OSError, ValueError):
        return defecto


def escribir_json(ruta, datos):
    tmp = ruta + ".tmp"
    with open(tmp, "w") as f:
        json.dump(datos, f, ensure_ascii=False, indent=1)
    os.replace(tmp, ruta)


# ---------------------------------------------------------------- Telegram

def _respuesta_http(abrir):
    try:
        with abrir() as r:
            return json.load(r)
    except urllib.error.HTTPError as e:
        try:
            return json.load(e)
        except Exception:
            return {"ok": False, "description": "HTTP %s" % e.code}
    except Exception as e:
        return {"ok": False, "description": type(e).__name__}


def tg(metodo, datos=None, timeout=70):
    url = "https://api.telegram.org/bot%s/%s" % (TOKEN, metodo)
    cuerpo = urllib.parse.urlencode(datos or {}).encode()
    return _respuesta_http(lambda: urllib.request.urlopen(url, cuerpo, timeout=timeout))


def decir(texto):
    log("→ telegram:", texto[:90].replace("\n", " "))
    for i in range(0, max(len(texto), 1), 3900):
        r = tg("sendMessage", {"chat_id": CHAT, "text": texto[i:i + 3900], "disable_web_page_preview": "true"})
        if not r.get("ok"):
            log("sendMessage falló:", r.get("description"))


def subir(metodo, campo, ruta, pie=""):
    frontera = uuid.uuid4().hex
    partes = []
    for k, v in (("chat_id", CHAT), ("caption", pie[:1000])):
        partes.append(('--%s\r\nContent-Disposition: form-data; name="%s"\r\n\r\n%s\r\n' % (frontera, k, v)).encode())
    with open(ruta, "rb") as f:
        contenido = f.read()
    cabecera = ('--%s\r\nContent-Disposition: form-data; name="%s"; filename="%s"\r\n'
                'Content-Type: application/octet-stream\r\n\r\n' % (frontera, campo, os.path.basename(ruta)))
    partes.append(cabecera.encode() + contenido + b"\r\n")
    partes.append(("--%s--\r\n" % frontera).encode())
    req = urllib.request.Request("https://api.telegram.org/bot%s/%s" % (TOKEN, metodo), b"".join(partes),
                                 {"Content-Type": "multipart/form-data; boundary=%s" % frontera})
    r = _respuesta_http(lambda: urllib.request.urlopen(req, timeout=900))
    if not r.get("ok"):
        log(metodo, "falló:", r.get("description"))
    return r

_enfriar = {}          # modelo -> instante hasta el que no se vuelve a intentar
_ultimo_bueno = [None]  # el último modelo que respondió va primero


def llm(mensajes, max_tokens=3000, presupuesto_s=180):
    """Prueba rutas gratuitas con nombre (y «auto» al final, rechazando Claude) dentro de un presupuesto de tiempo."""
    inicio = time.time()
    orden = sorted(RUTAS, key=lambda r: r[2] != _ultimo_bueno[0])
    for url, variable, modelo in orden:
        resto = presupuesto_s - (time.time() - inicio)
        if resto < 8:
            break
        if _enfriar.get(modelo, 0) > time.time() or not ENV.get(variable):
            continue
        cuerpo = json.dumps({"model": modelo, "messages": mensajes, "max_tokens": max_tokens,
                             "temperature": 0.2}).encode()
        req = urllib.request.Request(url, cuerpo, {"Authorization": "Bearer " + ENV[variable],
                                                   "Content-Type": "application/json"})
        d = _respuesta_http(lambda: urllib.request.urlopen(req, timeout=min(60, resto)))
        usado = str(d.get("model") or modelo)
        texto = ((d.get("choices") or [{}])[0].get("message") or {}).get("content") or ""
        texto = re.sub(r"<think>.*?</think>", "", texto, flags=re.S).strip()
        if texto and "claude" not in usado.lower():
            _ultimo_bueno[0] = modelo
            return texto, usado
        _enfriar[modelo] = time.time() + 600
        error = d.get("error")
        log("llm sin respuesta de", modelo, (error.get("message", "") if isinstance(error, dict) else d.get("description", ""))[:120])
    return "", ""

def por_reglas(texto, hay_encargo=False):
    t = texto.lower()
    artefactos = [{"tipo": k, "indicaciones": ""} for k, patron in CLAVES_TIPO if re.search(patron, t)]
    m = re.search(r"/d/([A-Za-z0-9_-]{20,})", texto)
    if m:
        fuente = "drive"
    elif re.search(r"s[ií]ntesis", t):
        fuente = "ultima_sintesis"
    else:
        fuente = "cuaderno"
    if re.search(r"recu[eé]rd(alo|a que|ame)|mi nombre es|ll[aá]mame", t):
        tipo = "preferencia"
    elif len(texto) > 400 or re.search(r"se aprueba|apruebo|desarroll|encarg|pr[oó]ximos pasos", t):
        tipo = "ajuste" if hay_encargo and len(texto) < 400 else "encargo"
    elif (hay_encargo and re.search(r"priori|añad|agreg|cambi|ajust|tambi[eé]n|en vez|quita|incluye|enf[oó]ca|"
                                    r"guion|storyboard|frente|personaje", t)
          and not re.search(r"env[ií]a|hazme|genera|m[aá]ndame", t)):
        tipo = "ajuste"
    else:
        tipo = "notebooklm" if artefactos else "respuesta"
    return {"tipo": tipo, "fuente": fuente, "drive_id": m.group(1) if m else "",
            "artefactos": artefactos if tipo == "notebooklm" else [], "recordar": "", "resumen": ""}


def interpretar(texto, hay_encargo=False):
    reglas = por_reglas(texto, hay_encargo)
    contexto = ("Hay un encargo en curso: %s." % hay_encargo) if hay_encargo else "No hay ningún encargo en curso."

                         {"role": "user", "content": texto}], 3000, 90)
    plan = None
    m = re.search(r"\{.*\}", crudo or "", re.S)
    if m:
        try:
            plan = json.loads(m.group(0))
        except ValueError:
            plan = None
    if not isinstance(plan, dict):
        log("interpretación por reglas (modelo no dio JSON)")
        return reglas
    artefactos = []
    for a in plan.get("artefactos") or []:
        if isinstance(a, dict) and a.get("tipo") in TIPOS and a["tipo"] not in [x["tipo"] for x in artefactos]:

    # Guardas: las reglas mandan en lo inequívoco (un encargo largo nunca se convierte en un audio).
    if reglas["tipo"] == "ajuste":
        tipo = "ajuste"
    elif reglas["tipo"] in ("encargo", "preferencia"):
        tipo = reglas["tipo"] if not (tipo == "ajuste" and hay_encargo and reglas["tipo"] == "encargo"
                                      and len(texto) < 400) else "ajuste"
    if tipo == "ajuste" and not hay_encargo:
        tipo = "respuesta"
    if tipo == "notebooklm" and not artefactos:
        artefactos = reglas["artefactos"] or []
        tipo = "notebooklm" if artefactos else "respuesta"
    fuente = plan.get("fuente") if plan.get("fuente") in ("ultima_sintesis", "drive", "cuaderno") else reglas["fuente"]

            "recordar": str(plan.get("recordar") or "")[:300], "resumen": str(plan.get("resumen") or "")[:300]}

def token_google(alias="alex"):
    c = leer_json(os.path.join(CASA, ".starseed", "cuentas", alias + ".json"), {})
    datos = urllib.parse.urlencode({"client_id": c.get("client_id", ""), "client_secret": c.get("client_secret", ""),
                                    "refresh_token": c.get("refresh_token", ""),
                                    "grant_type": "refresh_token"}).encode()
    d = _respuesta_http(lambda: urllib.request.urlopen(c.get("token_uri") or "https://oauth2.googleapis.com/token",
                                                       datos, timeout=30))
    if not d.get("access_token"):
        raise RuntimeError("no pude renovar el acceso a Drive de la cuenta %s (%s)" % (alias, d.get("error", "?")))
    return d["access_token"]


def drive(ruta, params):
    tok = token_google()
    url = "https://www.googleapis.com/drive/v3/%s?%s" % (ruta, urllib.parse.urlencode(params))
    req = urllib.request.Request(url, headers={"Authorization": "Bearer " + tok})
    return _respuesta_http(lambda: urllib.request.urlopen(req, timeout=30))


def ultima_sintesis():
    q = ("'%s' in parents and trashed=false and mimeType='application/vnd.google-apps.document'"
         % CARPETA_SINTESIS)
    d = drive("files", {"q": q, "orderBy": "createdTime desc", "pageSize": "30", "fields": "files(id,name)"})
    for a in d.get("files") or []:
        if re.match(r"^S[ií]ntesis diaria \d{4}-\d{2}-\d{2}", a.get("name", "")):
            return a["id"], a["name"]
    raise RuntimeError("no encontré ninguna «Síntesis diaria AAAA-MM-DD» en la carpeta de Drive")


def nombre_drive(fid):
    d = drive("files/" + fid, {"fields": "name"})
    if not d.get("name"):
        raise RuntimeError("no encuentro ese archivo de Drive con la cuenta alexbordongarrigos")
    return d["name"]


# ---------------------------------------------------------------- NotebookLM

nb_cerrojo = threading.Lock()  # una sola orden de NotebookLM a la vez: dos a la vez rotan la cookie y se pisan


def nb(args, timeout=600, esperar=True):
    """Ejecuta el CLI de NotebookLM. Devuelve (rc, stdout, stderr), o None si esperar=False y otra orden está en curso.
    Solo lanza SesionCaducada si la orden FALLA con un error de autenticación (los avisos con rc 0 no cuentan)."""
    if not nb_cerrojo.acquire(blocking=esperar):
        return None
    try:
        entorno = dict(os.environ, HOME=CASA, NOTEBOOKLM_HL="es")
        p = subprocess.run([NBLM] + args, capture_output=True, text=True, timeout=timeout, env=entorno, cwd=CASA)
    except subprocess.TimeoutExpired:
        return 124, "", "tiempo agotado"
    finally:
        nb_cerrojo.release()
    todo = (p.stdout or "") + (p.stderr or "")
    if p.returncode != 0 and re.search(r"Authentication (expired|error)|Run 'notebooklm login'", todo):
        raise SesionCaducada(todo[-300:])
    return p.returncode, p.stdout or "", p.stderr or ""


def json_de(texto):
    try:
        return json.loads(texto)
    except ValueError:
        m = re.search(r"\{.*\}", texto, re.S)
        if m:
            try:
                return json.loads(m.group(0))
            except ValueError:
                pass
    return {}


def sesion_viva():
    try:
        rc, _, _ = nb(["list", "--json"], 120)
        return rc == 0
    except SesionCaducada:
        return False


def asegurar_fuente(fid, titulo, cuaderno=CUADERNO):

    for s in json_de(out).get("sources") or []:
        if s.get("drive_document_id") == fid and s.get("status") == "ready":
            return s["id"]
    rc, out, err = nb(["source", "add-drive", "-n", cuaderno, fid, titulo, "--json"], 300)
    d = json_de(out)
    sid = d.get("source_id") or d.get("id") or (d.get("source") or {}).get("id")
    if rc != 0 or not sid:
        raise RuntimeError("no pude añadir «%s» al cuaderno: %s" % (titulo, (err or out)[-300:]))
    nb(["source", "wait", "-n", cuaderno, sid, "--timeout", "600"], 660)
    return sid


def lanzar(tipo, indicaciones, sid, cuaderno=CUADERNO):
    nombre, _ext, fijas, tope, idioma, espera = TIPOS[tipo]
    args = ["generate", tipo, "-n", cuaderno, "--json"] + fijas
    if sid:
        args += ["-s", sid]
    if idioma:
        args += ["--language", "es"]
    texto = indicaciones or INDICACIONES.get(tipo, "En español, con lo esencial de la fuente.")
    if not espera:  # mapa mental: es síncrono
        rc, out, err = nb(args + [texto], tope)
        if rc != 0:
            raise RuntimeError("%s: %s" % (nombre, (err or out)[-400:]))
        return "sincrono:" + json.dumps(json_de(out), ensure_ascii=False)
    rc, out, err = nb(args + ["--no-wait", "--retry", "2", texto], 300)
    tarea = json_de(out).get("task_id")
    if rc != 0 or not tarea:
        raise RuntimeError("%s no arrancó: %s" % (nombre, (err or out)[-400:]))
    return tarea


def esperar_y_bajar(tipo, tarea, peticion, cuaderno=CUADERNO):
    nombre, ext, _f, tope, _i, _e = TIPOS[tipo]
    os.makedirs(SALIDAS, exist_ok=True)
    ruta = os.path.join(SALIDAS, "%s-%s%s" % (peticion, tipo, ext))
    if tarea.startswith("sincrono:"):
        with open(ruta, "w") as f:
            f.write(tarea[len("sincrono:"):])
        return ruta
    rc, out, err = nb(["artifact", "wait", "-n", cuaderno, tarea, "--timeout", str(tope), "--json"], tope + 120)
    estado = json_de(out).get("status", "")
    if rc != 0 or estado not in ("completed", ""):
        raise RuntimeError("%s no terminó (%s): %s" % (nombre, estado or rc, (err or out)[-300:]))
    rc, out, err = nb(["download", tipo, "-n", cuaderno, "-a", tarea, ruta, "--force"], 900)
    if rc != 0 or not os.path.exists(ruta):
        raise RuntimeError("no pude descargar %s: %s" % (nombre, (err or out)[-300:]))
    return ruta


def ffmpeg(args):
    return subprocess.run(["ffmpeg", "-y", "-loglevel", "error"] + args, capture_output=True, timeout=1800).returncode


def enviar_resultado(tipo, ruta, titulo):
    nombre = TIPOS[tipo][0]
    pie = "%s · %s" % (nombre.capitalize(), titulo or "cuaderno StarSeed: Producción audiovisual")
    tam = os.path.getsize(ruta)
    if tipo == "infographic":
        if tam < 10 * 1024 * 1024:
            subir("sendPhoto", "photo", ruta, pie)
        subir("sendDocument", "document", ruta, pie + " (calidad completa)")
        return
    if tipo == "audio":
        mp3 = ruta[:-4] + ".mp3"
        if ffmpeg(["-i", ruta, "-vn", "-ac", "1", "-b:a", "96k", mp3]) == 0 and os.path.exists(mp3):
            if os.path.getsize(mp3) > 49 * 1024 * 1024:
                ffmpeg(["-i", ruta, "-vn", "-ac", "1", "-b:a", "48k", mp3])
            ruta = mp3
        r = subir("sendAudio", "audio", ruta, pie)
        if not r.get("ok"):
            subir("sendDocument", "document", ruta, pie)
        return
    if tipo == "video":
        if tam > 49 * 1024 * 1024:
            chico = ruta[:-4] + "-tg.mp4"
            if ffmpeg(["-i", ruta, "-vf", "scale=-2:720", "-c:v", "libx264", "-crf", "30", "-preset", "veryfast",
                       "-c:a", "aac", "-b:a", "96k", chico]) == 0:
                ruta = chico
        if os.path.getsize(ruta) <= 49 * 1024 * 1024:
            subir("sendVideo", "video", ruta, pie)
        else:
            decir("%s: pesa demasiado para Telegram; está en el cuaderno: %s" % (pie, ENLACE_CUADERNO))
        return
    subir("sendDocument", "document", ruta, pie)

# ---------------------------------------------------------------- Drive (escritura con alex-docs: drive.file)

def _drive_pedir(metodo, url, cuerpo, tipo):
    tok = token_google("alex-docs")
    req = urllib.request.Request(url, cuerpo, {"Authorization": "Bearer " + tok, "Content-Type": tipo}, method=metodo)
    return _respuesta_http(lambda: urllib.request.urlopen(req, timeout=600))


def drive_carpeta(nombre, padre):
    d = _drive_pedir("POST", "https://www.googleapis.com/drive/v3/files?fields=id,webViewLink",
                     json.dumps({"name": nombre, "mimeType": "application/vnd.google-apps.folder",
                                 "parents": [padre]}).encode(), "application/json")
    if not d.get("id"):
        raise RuntimeError("no pude crear la carpeta «%s» en Drive: %s" % (nombre, str(d)[:200]))
    return d["id"], d.get("webViewLink", "")


def drive_doc(nombre, markdown, padre=None, archivo_id=None):
    """Crea (o reescribe) un Google Doc a partir de markdown. Devuelve (id, enlace)."""
    meta = {"name": nombre}
    if not archivo_id:
        meta.update({"mimeType": "application/vnd.google-apps.document", "parents": [padre]})
    frontera = uuid.uuid4().hex
    cuerpo = ("--%s\r\nContent-Type: application/json; charset=UTF-8\r\n\r\n%s\r\n--%s\r\n"
              "Content-Type: text/markdown; charset=UTF-8\r\n\r\n" % (frontera, json.dumps(meta), frontera)).encode()
    cuerpo += markdown.encode("utf-8") + ("\r\n--%s--\r\n" % frontera).encode()
    url = "https://www.googleapis.com/upload/drive/v3/files%s?uploadType=multipart&fields=id,webViewLink" % (
        "/" + archivo_id if archivo_id else "")
    d = _drive_pedir("PATCH" if archivo_id else "POST", url, cuerpo, "multipart/related; boundary=" + frontera)
    if not d.get("id"):
        raise RuntimeError("no pude guardar «%s» en Drive: %s" % (nombre, str(d)[:200]))
    return d["id"], d.get("webViewLink", "")


def drive_archivo(ruta, padre, nombre=None):
    """Sube un archivo binario (png, mp3, mp4, pdf, csv, json) con subida reanudable. Devuelve (id, enlace)."""
    nombre = nombre or os.path.basename(ruta)
    tipos = {".png": "image/png", ".mp3": "audio/mpeg", ".m4a": "audio/mp4", ".mp4": "video/mp4",
             ".pdf": "application/pdf", ".csv": "text/csv", ".json": "application/json", ".md": "text/markdown"}
    mime = tipos.get(os.path.splitext(ruta)[1].lower(), "application/octet-stream")
    tok = token_google("alex-docs")
    req = urllib.request.Request(
        "https://www.googleapis.com/upload/drive/v3/files?uploadType=resumable&fields=id,webViewLink",
        json.dumps({"name": nombre, "parents": [padre]}).encode(),
        {"Authorization": "Bearer " + tok, "Content-Type": "application/json; charset=UTF-8",
         "X-Upload-Content-Type": mime}, method="POST")
    with urllib.request.urlopen(req, timeout=60) as r:
        destino = r.headers.get("Location")
    with open(ruta, "rb") as f:
        datos = f.read()
    req = urllib.request.Request(destino, datos, {"Content-Type": mime}, method="PUT")
    d = _respuesta_http(lambda: urllib.request.urlopen(req, timeout=1800))
    if not d.get("id"):
        raise RuntimeError("no pude subir %s a Drive: %s" % (nombre, str(d)[:200]))
    return d["id"], d.get("webViewLink", "")


# ---------------------------------------------------------------- encargos de varias horas (el productor)

ENTREGABLES_VALIDOS = ("informe", "documento") + tuple(TIPOS)
productor_trabajo = threading.Event()


def encargos():
    return leer_json(ENCARGOS, [])


def guardar_encargo(enc):
    """Guarda el encargo del productor sin perder lo que Maggasukha cambió entretanto (ajustes, pausa, cancelación)."""
    with bloqueo:
        todos = leer_json(ENCARGOS, [])
        viejo = next((e for e in todos if e["id"] == enc["id"]), None)
        if viejo:
            enc["notas"] = list(enc.get("notas", [])) + [n for n in viejo.get("notas", [])
                                                           if n not in enc.get("notas", [])]
            if viejo.get("estado") in ("pausado", "cancelado") and enc.get("estado") == "activo":
                enc["estado"] = viejo["estado"]
        todos = [e for e in todos if e["id"] != enc["id"]] + [enc]
        escribir_json(ENCARGOS, todos)


def modificar_encargo(eid, nota=None, estado=None):
    with bloqueo:
        todos = leer_json(ENCARGOS, [])
        for e in todos:
            if e["id"] == eid:
                if nota:
                    e.setdefault("notas", []).append(nota)
                if estado:
                    e["estado"] = estado
        escribir_json(ENCARGOS, todos)


def encargo_activo():
    for e in encargos():
        if e.get("estado") in ("activo", "pausado"):
            return e
    return None


def validar_plan(plan):
    """Normaliza un plan de encargo: ≤ 10 frentes, ≤ 3 investigaciones y ≤ 8 entregables por frente."""
    frentes = []
    for i, f in enumerate((plan.get("frentes") or [])[:10], 1):
        entregables = []
        for e in (f.get("entregables") or [])[:8]:
            if isinstance(e, dict) and e.get("tipo") in ENTREGABLES_VALIDOS and e.get("titulo"):
                entregables.append({"tipo": e["tipo"], "titulo": str(e["titulo"])[:150],
                                    "prompt": str(e.get("prompt") or e["titulo"])[:6000],
                                    "enviar": bool(e.get("enviar"))})
        if not entregables:
            continue
        frentes.append({"id": "F%d" % i, "nombre": str(f.get("nombre") or "Frente %d" % i)[:120],
                        "destino": f.get("destino") if f.get("destino") in DESTINOS else "produccion",
                        "investigacion": [str(q)[:1500] for q in (f.get("investigacion") or [])[:3]],
                        "fuentes_drive": [x for x in (f.get("fuentes_drive") or []) if isinstance(x, list)][:8],
                        "entregables": entregables, "estado": "pendiente"})
    if not frentes:
        raise ValueError("el plan no tiene frentes con entregables válidos")
    return frentes

def planificar(texto):
    crudo, modelo = llm([{"role": "system", "content": PROMPT_PLAN + " " + contexto_sistema()},
                         {"role": "user", "content": texto}], 6000, 240)
    m = re.search(r"\{.*\}", crudo or "", re.S)

    if m:

        except ValueError:
            plan = {}

        frentes = validar_plan({"frentes": [
            {"nombre": "Investigación", "destino": "fuentes", "investigacion": [texto[:1200]],
             "entregables": [{"tipo": "informe", "titulo": "Informe de investigación", "prompt": texto},
                             {"tipo": "mind-map", "titulo": "Mapa de la investigación"}]},
            {"nombre": "Desarrollo", "destino": "produccion",

            {"nombre": "Plan y listas de tareas", "destino": "produccion",
             "entregables": [{"tipo": "informe", "titulo": "Próximos pasos y listas de tareas", "prompt":
                              "Próximos pasos y listas de tareas por tipo de contenido para: " + texto}]}]})

def nuevo_encargo(texto, titulo, frentes, origen):
    enc = {"id": "ENC-" + datetime.datetime.now().strftime("%m%d-%H%M"), "titulo": titulo, "texto": texto,
           "frentes": frentes, "estado": "activo", "notas": [], "creado": time.time(), "origen": origen}

def texto_panel(enc):
    lineas = ["# Panel · %s" % enc["titulo"], "",
              "Encargo %s · estado: **%s** · actualizado %s" % (
                  enc["id"], enc["estado"], datetime.datetime.now().strftime("%Y-%m-%d %H:%M")), "",
              "> " + enc["texto"][:1500].replace("\n", "\n> "), ""]
    if enc.get("notas"):
        lineas += ["## Ajustes de Maggasukha", ""] + ["- " + n for n in enc["notas"]] + [""]
    for f in enc["frentes"]:
        marca = {"hecho": "✅", "en_curso": "⏳"}.get(f.get("estado"), "⬜")
        lineas.append("## %s %s · %s" % (marca, f["id"], f["nombre"]))
        if f.get("carpeta_enlace"):
            lineas.append("Carpeta: %s" % f["carpeta_enlace"])
        if f.get("cuaderno"):
            lineas.append("Cuaderno de NotebookLM: https://notebooklm.google.com/notebook/%s" % f["cuaderno"])
        for q in f.get("investigacion", []):
            lineas.append("- %s Investigación: %s" % ("✅" if q in f.get("investigado", []) else "⬜", q[:160]))
        for e in f["entregables"]:
            marca = "✅" if e.get("enlace") else ("⚠️" if e.get("fallo") else "⬜")
            extra = e.get("enlace") or e.get("fallo") or ""
            lineas.append("- %s %s · %s %s" % (marca, TIPOS.get(e["tipo"], (e["tipo"],))[0], e["titulo"], extra))
        lineas.append("")
    return "\n".join(lineas)


def actualizar_panel(enc):

        if enc.get("panel"):
            drive_doc("Panel · %s" % enc["titulo"], texto_panel(enc), archivo_id=enc["panel"])
        else:
            enc["panel"], enc["panel_enlace"] = drive_doc("Panel · %s (%s)" % (enc["titulo"], enc["id"]),
                                                          texto_panel(enc), CARPETA_PRODUCCION)
            guardar_encargo(enc)
    except Exception as e:
        log("panel:", e)


def cuaderno_de(frente, enc):
    if frente.get("cuaderno"):
        return frente["cuaderno"]
    rc, out, err = nb(["create", "StarSeed · %s · %s" % (enc["id"], frente["nombre"][:60]), "--json"], 180)
    d = json_de(out)
    cid = d.get("id") or (d.get("notebook") or {}).get("id") or d.get("active_notebook_id")
    if rc != 0 or not cid:
        raise RuntimeError("no pude crear el cuaderno de NotebookLM: %s" % (err or out)[-200:])
    frente["cuaderno"] = cid
    guardar_encargo(enc)
    for fid, titulo in frente.get("fuentes_drive", []):
        try:
            asegurar_fuente(fid, titulo, cid)
        except Exception as e:
            log("fuente base", titulo, e)
    return cid


def investigar(frente, enc, cid):
    for q in frente.get("investigacion", []):
        if q in frente.setdefault("investigado", []):
            continue
        rc, out, err = nb(["source", "add-research", "-n", cid, "--mode", "deep", "--import-all", "--cited-only",
                           "--timeout", "1500", q], 3300)
        if rc != 0:
            log("investigación falló:", (err or out)[-300:])
            frente.setdefault("avisos", []).append("una investigación no terminó: " + q[:80])
        frente["investigado"].append(q)
        guardar_encargo(enc)


def instrucciones(enc, e):
    notas = ("\nAjustes de Maggasukha que debes aplicar: " + " | ".join(enc["notas"])) if enc.get("notas") else ""
    return "%s\n\n%s%s\nEncargo original: %s" % (e["prompt"], REGLAS_MARCA, notas, enc["texto"][:2500])


def producir(e, frente, enc, cid):
    titulo = "%s · %s" % (e["titulo"], enc["id"])
    if e["tipo"] == "documento":
        rc, out, err = nb(["ask", "-n", cid, "--json", instrucciones(enc, e)], 600)
        d = json_de(out)
        respuesta = d.get("answer") or d.get("text") or ""
        if rc != 0 or not respuesta:
            raise RuntimeError("NotebookLM no respondió: %s" % (err or out)[-200:])
        return drive_doc(titulo, "# %s\n\n%s" % (e["titulo"], respuesta), frente["carpeta"])
    tipo = "report" if e["tipo"] == "informe" else e["tipo"]
    tarea = lanzar(tipo, instrucciones(enc, e), None, cid)
    ruta = esperar_y_bajar(tipo, tarea, "%s-%s" % (enc["id"], frente["id"]), cid)
    if tipo == "report":
        with open(ruta, encoding="utf-8") as f:
            return drive_doc(titulo, f.read(), frente["carpeta"])
    if e.get("enviar") or tipo == "infographic":
        try:
            enviar_resultado(tipo, ruta, e["titulo"])
        except Exception as ex:
            log("envío a Telegram:", ex)
    return drive_archivo(ruta, frente["carpeta"], titulo + os.path.splitext(ruta)[1])


def ejecutar_encargo(enc):
    if not enc.get("anunciado"):
        actualizar_panel(enc)
        decir("Encargo %s en marcha: «%s».\n%d frentes: %s.\nPanel con el avance: %s\nPuedes escribirme ajustes "
              "cuando quieras (los aplico en los frentes siguientes) · /pausa · /cancela" % (
                  enc["id"], enc["titulo"], len(enc["frentes"]),
                  "; ".join("%s %s" % (f["id"], f["nombre"]) for f in enc["frentes"]), enc.get("panel_enlace", "")))
        enc["anunciado"] = True
        guardar_encargo(enc)
    for frente in enc["frentes"]:
        actual = next((e for e in encargos() if e["id"] == enc["id"]), enc)
        enc["notas"], enc["estado"] = actual.get("notas", []), actual.get("estado", enc["estado"])
        if enc["estado"] in ("cancelado", "pausado"):
            return
        if frente.get("estado") == "hecho":
            continue
        frente["estado"] = "en_curso"
        if not frente.get("carpeta"):
            frente["carpeta"], frente["carpeta_enlace"] = drive_carpeta(
                "%s · %s (%s)" % (frente["id"], frente["nombre"], enc["id"]), DESTINOS[frente["destino"]])
        guardar_encargo(enc)
        cid = cuaderno_de(frente, enc)
        investigar(frente, enc, cid)
        for e in frente["entregables"]:
            if e.get("enlace"):
                continue
            try:
                _, e["enlace"] = producir(e, frente, enc, cid)
                e.pop("fallo", None)
            except SesionCaducada:
                raise
            except Exception as ex:
                e["fallo"] = str(ex)[:200]
                log("entregable falló", e["titulo"], traceback.format_exc()[-600:])
            guardar_encargo(enc)
        frente["estado"] = "hecho"
        guardar_encargo(enc)
        actualizar_panel(enc)
        hechos = [e for e in frente["entregables"] if e.get("enlace")]
        fallos = [e for e in frente["entregables"] if e.get("fallo")]
        n = sum(1 for f in enc["frentes"] if f.get("estado") == "hecho")
        decir("Avance %s · %d/%d · %s ✓\n%s%s\nCarpeta: %s" % (
            enc["id"], n, len(enc["frentes"]), frente["nombre"],
            "\n".join("• %s: %s" % (e["titulo"], e["enlace"]) for e in hechos),
            ("\nPendiente de reintentar: " + "; ".join(e["titulo"] for e in fallos)) if fallos else "",
            frente.get("carpeta_enlace", "")))
    pendientes = [e for f in enc["frentes"] for e in f["entregables"] if not e.get("enlace")]
    if pendientes and enc.get("vueltas", 0) < 2:
        enc["vueltas"] = enc.get("vueltas", 0) + 1
        for f in enc["frentes"]:
            if any(not e.get("enlace") for e in f["entregables"]):
                f["estado"] = "pendiente"
        guardar_encargo(enc)
        return
    enc["estado"] = "hecho"
    guardar_encargo(enc)
    actualizar_panel(enc)
    decir("Encargo %s terminado ✓ «%s»\n%d entregables en Drive%s.\nPanel: %s" % (
        enc["id"], enc["titulo"], sum(1 for f in enc["frentes"] for e in f["entregables"] if e.get("enlace")),
        (" · %d no salieron (detalle en el panel)" % len(pendientes)) if pendientes else "", enc.get("panel_enlace")))

def atender(p):
    actualizar(p["id"], estado="en_curso", intentos=p.get("intentos", 0) + 1)
    try:
        activo = encargo_activo()
        plan = p.get("plan") or interpretar(p["texto"], activo["titulo"] if activo else False)
        actualizar(p["id"], plan=plan)
        if plan["tipo"] == "preferencia":
            regla = plan.get("recordar") or p["texto"][:300]
            prefs = preferencias()
            if regla not in prefs:
                prefs.append(regla)
                escribir_json(PREFS, prefs)
            decir("Guardado en mis preferencias permanentes (A1): «%s». Lo aplico desde ya en todo lo que haga." % regla)
        elif plan["tipo"] == "ajuste" and activo:
            modificar_encargo(activo["id"], nota=p["texto"][:1500])
            decir("Anotado en el encargo %s ✓ Lo aplico en los frentes y entregables que faltan." % activo["id"])
        elif plan["tipo"] == "encargo":
            if activo:
                decir("Ya hay un encargo en curso (%s). Pongo este a continuación y lo planifico ahora." % activo["id"])
            titulo, frentes, modelo = planificar(p["texto"])
            enc = nuevo_encargo(p["texto"], titulo, frentes, "telegram")
            if activo:
                modificar_encargo(enc["id"], estado="en_espera")
            decir("Plan del encargo %s (%s): %d frentes, %d entregables. %s" % (
                enc["id"], modelo or "plan base", len(frentes), sum(len(f["entregables"]) for f in frentes),
                "Empiezo ya y te mando avances." if not activo else "Arranca al terminar el actual."))
        elif plan["tipo"] == "notebooklm" and plan["artefactos"]:
            hacer_notebooklm(p, plan)
        else:
            responder(p)
        actualizar(p["id"], estado="hecha", terminada=time.time())
    except SesionCaducada:
        time.sleep(30)
        if sesion_viva():  # fallo pasajero (cookie rotando): se reintenta sin molestar a Alex
            actualizar(p["id"], estado="reintentar")
            hay_trabajo.set()
            return
        actualizar(p["id"], estado="esperando_sesion")
        if not p.get("avisada_sesion"):
            actualizar(p["id"], avisada_sesion=True)
            decir("La sesión de NotebookLM del A1 caducó. Tu petición queda en cola y se hará sola en cuanto vuelva "
                  "(hasta 24 h). " + RENOVAR)
    except Exception as e:
        log("falló", p["id"], traceback.format_exc()[-1500:])
        n = p.get("intentos", 0) + 1
        if n < 3:
            actualizar(p["id"], estado="reintentar")
            decir("Un paso falló (%s). Lo reintento en 5 min (%d/3)." % (str(e)[:300], n))
        else:
            actualizar(p["id"], estado="fallida")
            decir("No pude completar tu petición tras 3 intentos: %s" % str(e)[:500])


def trabajar():
    while True:
        hay_trabajo.wait(timeout=REINTENTO_S)
        hay_trabajo.clear()
        with bloqueo:
            cola = leer_json(COLA, [])
        for p in cola:
            if p.get("estado") not in ("nueva", "en_curso", "reintentar", "esperando_sesion"):
                continue
            if time.time() - p.get("recibida", 0) > VIDA_S:
                actualizar(p["id"], estado="caducada")
                decir("Retiro de la cola tras 24 h sin poder hacerla: «%s»" % p["texto"][:120])
                continue
            if p.get("estado") == "esperando_sesion" and not sesion_viva():
                continue
            atender(p)


def _sesion_ok(orden, timeout):
    """True si la orden funciona o si otra orden de NotebookLM está en curso (la sesión se está usando)."""
    try:
        r = nb(orden, timeout, esperar=False)
        return r is None or r[0] == 0
    except SesionCaducada:
        return False


def mantener_sesion():
    """Refresca la sesión cada 20 min. Solo avisa tras 3 comprobaciones fallidas seguidas (unos 15 min), porque un
    fallo suelto suele ser una rotación de cookie a la vez que otra orden, no una caducidad."""
    avisado, fallos = False, 0
    while True:
        ok = _sesion_ok(["auth", "refresh"], 180)
        if not ok:
            time.sleep(60)
            ok = _sesion_ok(["list", "--json"], 120)
        fallos = 0 if ok else fallos + 1
        marca = {"viva": ok, "fallos_seguidos": fallos,
                 "comprobada": datetime.datetime.now().isoformat(timespec="minutes")}
        with bloqueo:
            escribir_json(ESTADO, dict(leer_json(ESTADO, {}), notebooklm=marca))
        if fallos >= 3 and not avisado:
            decir("Aviso de MetaGenesis: la sesión de NotebookLM del A1 caducó (3 comprobaciones seguidas). Las "
                  "peticiones esperan en cola. " + RENOVAR)
            avisado = True
        if ok and avisado:
            decir("La sesión de NotebookLM del A1 vuelve a estar viva ✓ Retomo lo que estaba en cola.")
            avisado = False
            hay_trabajo.set()
        time.sleep(REFRESCO_S if ok else 300)


def texto_estado():
    cola = leer_json(COLA, [])
    nb_estado = leer_json(ESTADO, {}).get("notebooklm", {})
    activas = [p for p in cola if p.get("estado") in ("nueva", "en_curso", "reintentar", "esperando_sesion")]
    lineas = ["Estado de MetaGenesis (A1):",
              "• NotebookLM: %s (comprobada %s)" % ("viva" if nb_estado.get("viva") else "caducada o sin comprobar",
                                                   nb_estado.get("comprobada", "—")),
              "• Peticiones en curso: %d · hechas: %d" % (len(activas),
                                                         len([p for p in cola if p.get("estado") == "hecha"]))]
    for p in activas[:5]:
        lineas.append("   – %s · %s" % (p["estado"], p["texto"][:70]))
    lineas.append(texto_encargos())
    return "\n".join(lineas)


def texto_encargos():
    todos = encargos()
    if not todos:
        return "• Encargos: ninguno."
    lineas = ["• Encargos:"]
    for e in todos[-5:]:
        hechos = sum(1 for f in e["frentes"] if f.get("estado") == "hecho")
        actual = next((f["nombre"] for f in e["frentes"] if f.get("estado") == "en_curso"), "")
        lineas.append("   – %s · %s · %s · frentes %d/%d%s%s" % (
            e["id"], e["estado"], e["titulo"][:60], hechos, len(e["frentes"]),
            (" · ahora: " + actual) if actual and e["estado"] == "activo" else "",
            ("\n     panel: " + e["panel_enlace"]) if e.get("panel_enlace") else ""))
    return "\n".join(lineas)


def recibir(upd):
    m = upd.get("message") or {}
    if str((m.get("chat") or {}).get("id")) != CHAT:
        log("ignoro un mensaje de otro chat")
        return
    texto = (m.get("text") or m.get("caption") or "").strip()
    if not texto:
        decir("Por ahora atiendo mensajes de texto.")
        return
    if texto.startswith("/estado"):
        decir(texto_estado())
        return
    if texto.startswith(("/start", "/ayuda", "/help")):
        decir(AYUDA)
        return
    if texto.startswith("/encargos"):
        decir(texto_encargos())
        return
    if texto.startswith(("/pausa", "/sigue", "/cancela")):
        activo = encargo_activo()
        if not activo:
            decir("No hay ningún encargo en curso.")
            return
        estado = {"/pausa": "pausado", "/sigue": "activo", "/cancela": "cancelado"}[texto.split()[0]]
        modificar_encargo(activo["id"], estado=estado)
        productor_trabajo.set()
        decir("Encargo %s: %s. %s" % (activo["id"], estado, "Lo hecho sigue en Drive." if estado != "activo"
                                      else "Sigo desde donde lo dejé."))
        return
    with bloqueo:
        cola = leer_json(COLA, [])
        if any(p["id"] == upd["update_id"] for p in cola):
            return
        cola.append({"id": upd["update_id"], "texto": texto, "recibida": time.time(), "estado": "nueva",
                     "intentos": 0})
        escribir_json(COLA, cola[-200:])
    decir("Recibido ✓")
    hay_trabajo.set()


def escuchar():
    offset = leer_json(ESTADO, {}).get("offset", 0)
    while True:
        r = tg("getUpdates", {"timeout": 50, "offset": offset, "allowed_updates": json.dumps(["message"])}, 75)
        if not r.get("ok"):
            log("getUpdates:", r.get("description"))
            time.sleep(15)
            continue
        for u in r.get("result") or []:
            try:
                recibir(u)
            except Exception:
                log("recibir falló", traceback.format_exc()[-800:])
            offset = u["update_id"] + 1
            with bloqueo:
                escribir_json(ESTADO, dict(leer_json(ESTADO, {}), offset=offset))


def main():
    if not TOKEN or not CHAT:
        raise SystemExit("faltan TELEGRAM_SINTESIS_BOT_TOKEN o TELEGRAM_SINTESIS_CHAT_ID en ~/.starseed/env")
    os.makedirs(SALIDAS, exist_ok=True)
    for x in leer_json(COLA, []):  # tras un reinicio, lo que quedó a medias se retoma
        if x.get("estado") == "en_curso":
            actualizar(x["id"], estado="reintentar")
    preferencias()
    log("escuchando @StarSeed_Design_bot")
    threading.Thread(target=trabajar, daemon=True).start()
    threading.Thread(target=mantener_sesion, daemon=True).start()
    threading.Thread(target=productor, daemon=True).start()
    hay_trabajo.set()
    productor_trabajo.set()
    escuchar()


def encargar_desde_archivo(ruta):
    """CLI: deja un encargo ya planificado (por la dirección: Claude o un director) para que el servicio lo ejecute."""
    with open(ruta, encoding="utf-8") as f:
        plan = json.load(f)
    frentes = validar_plan(plan)
    enc = {"id": plan.get("id") or "ENC-" + datetime.datetime.now().strftime("%m%d-%H%M"),
           "titulo": plan["titulo"], "texto": plan["texto"], "frentes": frentes, "notas": plan.get("notas", []),
           "estado": "en_espera" if encargo_activo() else "activo", "creado": time.time(),
           "origen": plan.get("origen", "dirección")}
    guardar_encargo(enc)
    print("encargo", enc["id"], enc["estado"], len(frentes), "frentes",
          sum(len(f["entregables"]) for f in frentes), "entregables")
